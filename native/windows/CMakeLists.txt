cmake_minimum_required(VERSION 3.20)
project(chdss_audio_windows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include(CTest)
add_executable(policy-tests "${REPO_ROOT}/test/native-windows-policy.cpp")
add_test(NAME windows-audio-policy COMMAND policy-tests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "chdss-audio requires Windows x64")
  endif()
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
    message(FATAL_ERROR "Use an x64 generator/toolchain, not ARM64")
  endif()
  add_executable(chdss-audio main.cpp)
  add_executable(chdss-test-tone "${REPO_ROOT}/test/native-windows-tone.cpp")
  foreach(target chdss-audio chdss-test-tone)
    target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(${target} PRIVATE ole32 uuid mmdevapi)
  endforeach()
  set(OUTPUT_DIR "${REPO_ROOT}/native/bin/win32-x64")
  foreach(target chdss-audio policy-tests chdss-test-tone)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIR}" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${OUTPUT_DIR}" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${OUTPUT_DIR}")
  endforeach()
endif()
foreach(target policy-tests chdss-audio chdss-test-tone)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /WX /EHsc /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
      if(MINGW)
        target_link_options(${target} PRIVATE -static)
      endif()
    endif()
  endif()
endforeach()
